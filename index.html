async function loadStage(){ try { const res = await fetch("./stage.html"); if (!res.ok) { // Datei fehlt → 404 laden const err = await fetch("./404.html"); const errHtml = await err.text(); document.getElementById("stagePanel").innerHTML = errHtml; return; } const html = await res.text(); document.getElementById("stagePanel").innerHTML = html; } catch (e) { // Fehler → 404 laden const err = await fetch("./404.html"); const errHtml = await err.text(); document.getElementById("stagePanel").innerHTML = errHtml; } }
